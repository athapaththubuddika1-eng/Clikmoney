import express from "express"; import fetch from "node-fetch"; import dotenv from "dotenv"; import cors from "cors"; import { MongoClient, ObjectId } from "mongodb"; import bodyParser from "body-parser"; dotenv.config(); const app = express(); app.use(cors()); app.use(bodyParser.json()); const BOT_TOKEN = process.env.BOT_TOKEN; const ADMIN_CHAT_ID = process.env.ADMIN_CHAT_ID; // -5419054691 const MONGO_URI = process.env.MONGO_URI; if (!BOT_TOKEN || !ADMIN_CHAT_ID || !MONGO_URI) { console.error("Missing env vars: BOT_TOKEN, ADMIN_CHAT_ID, MONGO_URI"); } const TELEGRAM_API = (path) => `https://api.telegram.org/bot${BOT_TOKEN}/${path}`; const client = new MongoClient(MONGO_URI, {}); let db, usersCol, withdrawsCol, earningsCol; async function connectDB() { await client.connect(); db = client.db("clickmoney"); usersCol = db.collection("users"); withdrawsCol = db.collection("withdraws"); earningsCol = db.collection("earnings"); console.log("✅ MongoDB connected"); } connectDB().catch((e) => console.error("DB connect error", e)); /** Helper: send message to telegram (text + optional reply_markup) */ async function sendTelegramMessage(chat_id, text, reply_markup) { try { const body = { chat_id, text, parse_mode: "Markdown" }; if (reply_markup) body.reply_markup = reply_markup; const res = await fetch(TELEGRAM_API("sendMessage"), { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body), }); return res.json(); } catch (err) { console.error("sendTelegramMessage error:", err); } } /** Helper: edit message text (used to update Approve/Reject UI) */ async function editTelegramMessage(chat_id, message_id, text) { try { await fetch(TELEGRAM_API("editMessageText"), { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id, message_id, text, parse_mode: "Markdown" }), }); } catch (e) { console.error("editTelegramMessage error:", e); } } /** --- GET balance for user --- */ app.get("/api/balance/:userId", async (req, res) => { try { const userId = req.params.userId; const user = await usersCol.findOne({ userId }); res.json({ balance: user?.balance || 0, referrals: user?.referrals || 0, referrerId: user?.referrerId || null }); } catch (e) { console.error(e); res.status(500).json({ error: "Server error" }); } }); /** --- Referral registration: body { referrerId, newUserId } --- */ app.post("/api/referral", async (req, res) => { try { const { referrerId, newUserId } = req.body; if (!referrerId || !newUserId) return res.status(400).json({ error: "Missing data" }); // if new user exists, skip const existing = await usersCol.findOne({ userId: newUserId }); if (existing) return res.json({ message: "Already registered" }); // create new user with referrerId await usersCol.insertOne({ userId: newUserId, balance: 0, referrals: 0, referrerId: referrerId }); // give referrer $0.01 and increment referrals await usersCol.updateOne({ userId: referrerId }, { $inc: { balance: 0.01, referrals: 1 } }, { upsert: true }); // notify admin/channel const text = `🎁 *Referral Bonus!*\n👥 Referrer: ${referrerId}\n🆕 New User: ${newUserId}\n💰 Bonus: $0.01 + 10% commission`; await sendTelegramMessage(ADMIN_CHAT_ID, text); res.json({ success: true }); } catch (e) { console.error(e); res.status(500).json({ error: "Server error" }); } }); /** --- Earn endpoint (ads or captcha): { userId, type, amount } --- */ app.post("/api/earn", async (req, res) => { try { const { userId, type, amount } = req.body; if (!userId || !type || !amount) return res.status(400).json({ error: "Missing data" }); const amt = parseFloat(amount); if (isNaN(amt) || amt <= 0) return res.status(400).json({ error: "Invalid amount" }); // credit user await usersCol.updateOne({ userId }, { $inc: { balance: amt } }, { upsert: true }); // log earning await earningsCol.insertOne({ userId, type, amount: amt, date: new Date() }); // give 10% commission to referrer if exists const user = await usersCol.findOne({ userId }); const referrerId = user?.referrerId; if (referrerId) { const commission = +(amt * 0.10).toFixed(8); if (commission > 0) { await usersCol.updateOne({ userId: referrerId }, { $inc: { balance: commission } }, { upsert: true }); const commissionText = `🔁 *Referral Commission*\n👥 Referrer: ${referrerId}\n👤 From User: ${userId}\n💰 Commission: $${commission.toFixed(8)}`; await sendTelegramMessage(ADMIN_CHAT_ID, commissionText); } } const text = `🪙 *Earning Activity*\n👤 User: ${userId}\n📘 Type: ${type}\n💰 Amount: $${amt.toFixed(8)}`; await sendTelegramMessage(ADMIN_CHAT_ID, text); res.json({ success: true }); } catch (e) { console.error(e); res.status(500).json({ error: "Server error" }); } }); /** --- Withdraw request: body { wallet, amount, userId } --- */ app.post("/api/withdraw", async (req, res) => { try { const { wallet, amount, userId } = req.body; if (!wallet || !amount || !userId) return res.status(400).json({ error: "Missing data" }); const amt = parseFloat(amount); if (isNaN(amt) || amt <= 0) return res.status(400).json({ error: "Invalid amount" }); const user = await usersCol.findOne({ userId }); if (!user || (user.balance || 0) < amt) return res.status(400).json({ error: "Insufficient balance" }); // deduct balance await usersCol.updateOne({ userId }, { $inc: { balance: -amt } }); // create withdraw doc const withdrawDoc = { userId, wallet, amount: amt, status: "pending", createdAt: new Date(), }; const insertR = await withdrawsCol.insertOne(withdrawDoc); const withdrawId = insertR.insertedId.toString(); // send admin message with inline approve/reject buttons const text = `💸 *Withdraw Request*\n👤 User: ${userId}\n💰 Amount: $${amt.toFixed(2)}\n🏦 Wallet: \`${wallet}\`\n\nID: \`${withdrawId}\``; const reply_markup = { inline_keyboard: [ [ { text: "✅ Approve", callback_data: `approve_${withdrawId}` }, { text: "❌ Reject", callback_data: `reject_${withdrawId}` }, ], ], }; const tgRes = await sendTelegramMessage(ADMIN_CHAT_ID, text, reply_markup); // store telegram message identifiers so we can edit the message later try { const tgMessage = tgRes.result; await withdrawsCol.updateOne({ _id: insertR.insertedId }, { $set: { tg_chat_id: tgMessage.chat.id, tg_message_id: tgMessage.message_id } }); } catch (e) { // ignore if can't store tg message info } res.json({ success: true, withdrawId }); } catch (e) { console.error(e); res.status(500).json({ error: "Server error" }); } }); /** --- Get withdraws for a user (history) --- */ app.get("/api/withdraws/:userId", async (req, res) => { try { const userId = req.params.userId; const items = await withdrawsCol.find({ userId }).sort({ createdAt: -1 }).toArray(); res.json({ withdraws: items }); } catch (e) { console.error(e); res.status(500).json({ error: "Server error" }); } }); /** --- Telegram webhook endpoint for callback_query (approve/reject) --- * Telegram should be configured to send updates to /api/webhook * (setWebhook to https:///api/webhook) */ app.post("/api/webhook", async (req, res) => { try { const body = req.body; // handle callback_query (Approve/Reject) if (body.callback_query) { const cb = body.callback_query; const data = cb.data; // e.g. "approve_" or "reject_" const from = cb.from; // admin user const message = cb.message; if (!data) return res.sendStatus(200); const [action, withdrawId] = data.split("_"); if (!withdrawId) return res.sendStatus(200); const _id = ObjectId.isValid(withdrawId) ? new ObjectId(withdrawId) : null; if (!_id) return res.sendStatus(200); const w = await withdrawsCol.findOne({ _id }); if (!w) { // answer callback await fetch(TELEGRAM_API("answerCallbackQuery"), { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ callback_query_id: cb.id, text: "Withdraw not found." }) }); return res.sendStatus(200); } if (action === "approve") { await withdrawsCol.updateOne({ _id }, { $set: { status: "approved", processedAt: new Date(), processedBy: from.id } }); // edit admin message const newText = `💸 *Withdraw Request — APPROVED*\n👤 User: ${w.userId}\n💰 Amount: $${w.amount.toFixed(2)}\n🏦 Wallet: \`${w.wallet}\`\n\nID: \`${withdrawId}\``; if (message) await editTelegramMessage(message.chat.id, message.message_id, newText); // notify user (if possible) via chat id equal to userId (if numeric) else skip try { await sendTelegramMessage(w.userId, `✅ Your withdraw request of $${w.amount.toFixed(2)} has been *APPROVED*. Please wait for manual payout.` ); } catch (e) { /* ignore */ } // answer callback await fetch(TELEGRAM_API("answerCallbackQuery"), { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ callback_query_id: cb.id, text: "Approved" }) }); } else if (action === "reject") { // on reject, set status rejected and refund amount to user await withdrawsCol.updateOne({ _id }, { $set: { status: "rejected", processedAt: new Date(), processedBy: from.id } }); // refund await usersCol.updateOne({ userId: w.userId }, { $inc: { balance: w.amount } }, { upsert: true }); if (message) { const newText = `💸 *Withdraw Request — REJECTED*\n👤 User: ${w.userId}\n💰 Amount: $${w.amount.toFixed(2)}\n🏦 Wallet: \`${w.wallet}\`\n\nID: \`${withdrawId}\``; await editTelegramMessage(message.chat.id, message.message_id, newText); } try { await sendTelegramMessage(w.userId, `❌ Your withdraw request of $${w.amount.toFixed(2)} has been *REJECTED*. Amount refunded to your balance.` ); } catch (e) { /* ignore */ } await fetch(TELEGRAM_API("answerCallbackQuery"), { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ callback_query_id: cb.id, text: "Rejected & refunded" }) }); } return res.sendStatus(200); } // else ignore other update types res.sendStatus(200); } catch (e) { console.error("webhook error", e); res.sendStatus(500); } }); /** health */ app.get("/api/health", (req, res) => res.json({ ok: true })); // export app for Vercel export default app;
